			deity:omen_washington = {
				holy_site_deity_check_trigger = yes
			}
		}
	}

	passive_modifier = { land_morale_modifier = deity_land_morale_modifier_svalue }

		can_deify_trigger = yes

		military_apotheosis_military_experience_effect = yes
	}
}
